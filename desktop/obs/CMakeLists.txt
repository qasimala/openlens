find_package(PkgConfig REQUIRED)
pkg_check_modules(OBSDEPS REQUIRED IMPORTED_TARGET libobs)
if(OBSDEPS_FOUND)
  add_library(openlens-obs MODULE src/openlens_obs.cpp)
  target_link_libraries(openlens-obs PRIVATE openlens_core ${OBSDEPS_LINK_LIBRARIES})
  target_compile_features(openlens-obs PRIVATE cxx_std_20)
  set_target_properties(openlens-obs PROPERTIES PREFIX "" OUTPUT_NAME "openlens-obs")
  if(OPENLENS_BUILD_TESTS)
    add_executable(obs_plugin_smoke tests/obs_plugin_smoke.cpp)
    target_link_libraries(obs_plugin_smoke PRIVATE ${OBSDEPS_LINK_LIBRARIES})
    add_test(NAME obs_plugin_smoke
      COMMAND obs_plugin_smoke $<TARGET_FILE:openlens-obs>)
    set_tests_properties(obs_plugin_smoke PROPERTIES SKIP_RETURN_CODE 77)
  endif()
else()
  message(WARNING "libobs development files not found; native OBS plugin disabled")
endif()
